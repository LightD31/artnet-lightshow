# Python dependencies for src/analysis/ (the audio analysis pipeline the
# auto-show runs, launched via src/analyze.py). Install with:
#
#     pip install -r requirements.txt
#
# Then fetch the genre-classifier model once (~310 MB):
#
#     python scripts/setup-panns.py
#     python scripts/setup-panns.py --check     # verify without downloading
#
# ffmpeg must also be on PATH — src/deezer.js shells out to it to convert
# downloaded audio to WAV. It is not installable from pip; use your package
# manager (apt install ffmpeg / brew install ffmpeg / winget install ffmpeg).

# ── Core analysis ───────────────────────────────────────────────────────────
librosa>=0.10
numpy>=1.24
scipy>=1.10
scikit-learn>=1.3
soundfile>=0.12

# Used to pin BLAS thread counts around the parallel analysis stages.
threadpoolctl>=3.1

# ── Genre classification (PANNs) ────────────────────────────────────────────
# Without these the analyser still runs, but src/analysis/tagger.py catches
# the ImportError and skips silently — palettes and patterns then fall back to
# the mood/circumplex path and shows are quietly worse with no error shown.
#
# torch is deliberately not pinned to an exact version: the right wheel depends
# on platform and CUDA. If pip picks a build you don't want, install torch
# yourself first (see https://pytorch.org/get-started/locally/) and then run
# this file.
torch>=2.0
panns_inference>=0.1.1

# ── Audio download fallback ─────────────────────────────────────────────────
# Used when a track has no ISRC or Deezer is not configured.
yt-dlp>=2024.1.1
